# Create empty array initialised to 0
cooc_matrix <- matrix(0, nrow = length(all_species), ncol = length(all_species),
                      dimnames = list(all_species, all_species))

# Fill the matrix with the values of the number of shared grids
for (i in 1:nrow(cooc_df)) {
  sp1 <- cooc_df$sp1[i]
  sp2 <- cooc_df$sp2[i]
  val <- cooc_df$cooccurrence_cells[i]
  
  if (sp1 %in% all_species && sp2 %in% all_species) {
    cooc_matrix[sp1, sp2] <- val
    cooc_matrix[sp2, sp1] <- val
  } else {
    warning(paste("Species not found:", sp1, sp2))
  }
}

MENTEL TEST

# We convert to ‘dist’ objects
jaccard_dist <- as.dist(1 - jaccard_sub)  # converts similarity to distance
cooc_dist <- as.dist(cooc_sub)            # is now a valid measure (number of shared cells)

# Mantel test
mantel_result <- mantel(cooc_dist, jaccard_dist, method = "pearson", permutations = 999)

# Result
print(mantel_result)
## 
## Mantel statistic based on Pearson's product-moment correlation 
## 
## Call:
## mantel(xdis = cooc_dist, ydis = jaccard_dist, method = "pearson",      permutations = 999) 
## 
## Mantel statistic r: -0.8301 
##       Significance: 1 
## 
## Upper quantiles of permutations (null model):
##   90%   95% 97.5%   99% 
## 0.107 0.133 0.155 0.168 
## Permutation: free
## Number of permutations: 999

r = -0.83 indicates a very strong negative correlation between the two matrices.

The matrices are:

  • cooc_matrix: number of cells shared between pairs of species.

  • jaccard_sub: niche similarity, converted to distance (1 - Jaccard).

Therefore:

  • The greater the niche distance (the less similar the niches are), the greater the co-occurrence.

  • This contradicts the typical ecological expectation (that species with more similar niches co-occur more).
